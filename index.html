curl -fsSL https://ismaelrajaomaniry.github.io/pos-suite-desktop/apt/pos-suite-archive-keyring.asc \
  | sudo gpg --dearmor -o /usr/share/keyrings/pos-suite-archive-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/pos-suite-archive-keyring.gpg] https://ismaelrajaomaniry.github.io/pos-suite-desktop/apt stable main" \
  | sudo tee /etc/apt/sources.list.d/pos-suite.list
sudo apt update && sudo apt install pos-suite-serveur
